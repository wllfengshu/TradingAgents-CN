# 因子有效性评价体系

## 一、核心评价指标

### 1. IC (Information Coefficient) - 信息系数

```python
import pandas as pd
import numpy as np
from scipy import stats

def calculate_ic(factor_values, forward_returns):
    """
    计算IC值 - 因子值与未来收益的相关性
    
    Parameters:
    -----------
    factor_values: pd.Series - 截面因子值
    forward_returns: pd.Series - 未来N期收益率
    
    Returns:
    --------
    ic: float - Pearson相关系数
    """
    # Pearson IC
    ic, p_value = stats.pearsonr(factor_values, forward_returns)
    return ic, p_value

def calculate_rank_ic(factor_values, forward_returns):
    """
    计算Rank IC - 使用Spearman秩相关
    更鲁棒，对异常值不敏感（实际更常用）
    """
    rank_ic, p_value = stats.spearmanr(factor_values, forward_returns)
    return rank_ic, p_value
```

### 2. ICIR (IC Information Ratio) - IC信息比率

```python
def calculate_icir(ic_series):
    """
    ICIR = Mean(IC) / Std(IC)
    
    衡量IC的稳定性，类似夏普比率
    
    Parameters:
    -----------
    ic_series: pd.Series - 每期IC值的时间序列
    """
    mean_ic = ic_series.mean()
    std_ic = ic_series.std()
    
    icir = mean_ic / std_ic if std_ic != 0 else 0
    
    return icir

# 评价标准
"""
IC:
  |IC| > 0.02  : 弱有效
  |IC| > 0.05  : 中等有效  
  |IC| > 0.10  : 强有效

ICIR:
  |ICIR| > 0.5 : 可接受
  |ICIR| > 1.0 : 良好
  |ICIR| > 2.0 : 优秀
"""
```

## 二、完整因子评价框架

```python
class FactorEvaluator:
    """
    全面的因子有效性评价系统
    """
    
    def __init__(self, factor_data, price_data, periods=[1, 5, 10, 20]):
        """
        Parameters:
        -----------
        factor_data  : pd.DataFrame - index=date, columns=stock_code
        price_data   : pd.DataFrame - index=date, columns=stock_code  
        periods      : list - 预测周期(交易日)
        """
        self.factor_data = factor_data
        self.price_data = price_data
        self.periods = periods
        self.results = {}
    
    # ==================== IC系列 ====================
    
    def calc_ic_series(self, period=5):
        """计算每期截面IC时间序列"""
        forward_returns = self.price_data.pct_change(period).shift(-period)
        
        ic_list = []
        rank_ic_list = []
        
        for date in self.factor_data.index:
            if date not in forward_returns.index:
                continue
                
            factor = self.factor_data.loc[date].dropna()
            ret = forward_returns.loc[date].dropna()
            
            # 取交集
            common = factor.index.intersection(ret.index)
            if len(common) < 10:  # 样本太少跳过
                continue
                
            f = factor[common]
            r = ret[common]
            
            ic, _ = stats.pearsonr(f, r)
            rank_ic, _ = stats.spearmanr(f, r)
            
            ic_list.append({'date': date, 'IC': ic, 'Rank_IC': rank_ic})
        
        ic_df = pd.DataFrame(ic_list).set_index('date')
        return ic_df
    
    def calc_ic_summary(self, ic_series):
        """IC统计摘要"""
        summary = {
            'IC_Mean':       ic_series['IC'].mean(),
            'IC_Std':        ic_series['IC'].std(),
            'ICIR':          ic_series['IC'].mean() / ic_series['IC'].std(),
            'IC_Positive_Ratio': (ic_series['IC'] > 0).mean(),  # IC>0占比
            
            'Rank_IC_Mean':  ic_series['Rank_IC'].mean(),
            'Rank_IC_Std':   ic_series['Rank_IC'].std(),
            'Rank_ICIR':     ic_series['Rank_IC'].mean() / ic_series['Rank_IC'].std(),
            
            # t检验 - IC均值是否显著异于0
            'IC_t_stat':     stats.ttest_1samp(ic_series['IC'].dropna(), 0)[0],
            'IC_p_value':    stats.ttest_1samp(ic_series['IC'].dropna(), 0)[1],
        }
        return pd.Series(summary)
    
    # ==================== 分层回测 ====================
    
    def calc_quantile_returns(self, period=5, n_quantiles=5):
        """
        分层收益分析 - 最直观的因子有效性验证
        将股票按因子值分成N组，看收益是否单调
        """
        forward_returns = self.price_data.pct_change(period).shift(-period)
        
        quantile_returns = {i: [] for i in range(1, n_quantiles + 1)}
        
        for date in self.factor_data.index:
            if date not in forward_returns.index:
                continue
                
            factor = self.factor_data.loc[date].dropna()
            ret = forward_returns.loc[date].dropna()
            common = factor.index.intersection(ret.index)
            
            if len(common) < n_quantiles * 5:
                continue
            
            f = factor[common]
            r = ret[common]
            
            # 分组
            quantiles = pd.qcut(f, n_quantiles, labels=False) + 1
            
            for q in range(1, n_quantiles + 1):
                mask = quantiles == q
                quantile_returns[q].append(r[mask].mean())
        
        # 转为DataFrame
        result = pd.DataFrame({
            f'Q{q}': quantile_returns[q] 
            for q in range(1, n_quantiles + 1)
        })
        
        return result
    
    def calc_long_short_return(self, quantile_returns_df):
        """
        多空组合收益 = Q5 - Q1
        核心指标：多头超额 + 空头超额
        """
        ls_return = quantile_returns_df['Q5'] - quantile_returns_df['Q1']
        
        summary = {
            'LS_Mean_Return':   ls_return.mean(),
            'LS_Std_Return':    ls_return.std(),
            'LS_Sharpe':        ls_return.mean() / ls_return.std() * np.sqrt(252),
            'LS_Win_Rate':      (ls_return > 0).mean(),
            'LS_Max_Drawdown':  self._max_drawdown(ls_return.cumsum()),
        }
        return summary
    
    # ==================== 因子衰减分析 ====================
    
    def calc_factor_decay(self, max_period=20):
        """
        因子IC随预测期的衰减
        判断因子的有效预测窗口
        """
        decay_results = []
        
        for period in range(1, max_period + 1):
            ic_series = self.calc_ic_series(period=period)
            summary = self.calc_ic_summary(ic_series)
            
            decay_results.append({
                'period':    period,
                'IC_Mean':   summary['IC_Mean'],
                'Rank_ICIR': summary['Rank_ICIR'],
            })
        
        return pd.DataFrame(decay_results).set_index('period')
    
    # ==================== 因子自相关 ====================
    
    def calc_factor_autocorr(self, lag=1):
        """
        因子自相关性
        高自相关 -> 换手率低 -> 交易成本低
        低自相关 -> 需要频繁调仓
        """
        autocorr_list = []
        
        dates = sorted(self.factor_data.index)
        
        for i in range(lag, len(dates)):
            curr_date = dates[i]
            prev_date = dates[i - lag]
            
            curr_factor = self.factor_data.loc[curr_date].dropna()
            prev_factor = self.factor_data.loc[prev_date].dropna()
            
            common = curr_factor.index.intersection(prev_factor.index)
            if len(common) < 10:
                continue
            
            corr, _ = stats.spearmanr(curr_factor[common], prev_factor[common])
            autocorr_list.append(corr)
        
        return {
            'mean_autocorr': np.mean(autocorr_list),
            'autocorr_series': autocorr_list
        }
    
    # ==================== 辅助函数 ====================
    
    def _max_drawdown(self, cumulative_returns):
        """最大回撤"""
        rolling_max = cumulative_returns.cummax()
        drawdown = cumulative_returns - rolling_max
        return drawdown.min()
    
    # ==================== 综合评分 ====================
    
    def comprehensive_score(self, period=5):
        """
        综合因子评分 (0-100)
        """
        ic_series = self.calc_ic_series(period)
        summary = self.calc_ic_summary(ic_series)
        
        scores = {}
        total_score = 0
        
        # 1. IC均值 (25分)
        ic_mean = abs(summary['IC_Mean'])
        if ic_mean >= 0.10:   scores['IC_Score'] = 25
        elif ic_mean >= 0.05: scores['IC_Score'] = 15
        elif ic_mean >= 0.02: scores['IC_Score'] = 8
        else:                  scores['IC_Score'] = 0
        
        # 2. ICIR (25分)
        icir = abs(summary['Rank_ICIR'])
        if icir >= 2.0:   scores['ICIR_Score'] = 25
        elif icir >= 1.0: scores['ICIR_Score'] = 15
        elif icir >= 0.5: scores['ICIR_Score'] = 8
        else:              scores['ICIR_Score'] = 0
        
        # 3. IC胜率 (25分)
        win_rate = summary['IC_Positive_Ratio']
        win_rate = win_rate if summary['IC_Mean'] > 0 else (1 - win_rate)
        if win_rate >= 0.60:   scores['WinRate_Score'] = 25
        elif win_rate >= 0.55: scores['WinRate_Score'] = 15
        elif win_rate >= 0.50: scores['WinRate_Score'] = 8
        else:                   scores['WinRate_Score'] = 0
        
        # 4. 统计显著性 (25分)
        p_value = summary['IC_p_value']
        if p_value <= 0.01:   scores['Significance_Score'] = 25
        elif p_value <= 0.05: scores['Significance_Score'] = 15
        elif p_value <= 0.10: scores['Significance_Score'] = 8
        else:                  scores['Significance_Score'] = 0
        
        total_score = sum(scores.values())
        scores['Total_Score'] = total_score
        scores['Grade'] = self._get_grade(total_score)
        
        return scores
    
    def _get_grade(self, score):
        if score >= 80: return 'A - 优秀因子'
        elif score >= 60: return 'B - 良好因子'
        elif score >= 40: return 'C - 一般因子'
        else: return 'D - 无效因子'
```

## 四、可视化分析

```python
import matplotlib.pyplot as plt
import seaborn as sns

def plot_factor_analysis(ic_series, quantile_returns):
    
    fig, axes = plt.subplots(2, 2, figsize=(15, 10))
    
    # 1. IC时间序列
    ax1 = axes[0, 0]
    ic_series['Rank_IC'].plot(ax=ax1, alpha=0.7)
    ic_series['Rank_IC'].rolling(12).mean().plot(ax=ax1, color='red', linewidth=2)
    ax1.axhline(y=0, color='black', linestyle='--')
    ax1.set_title('Rank IC 时间序列')
    ax1.legend(['Rank IC', '12期移动均值'])
    
    # 2. IC分布直方图
    ax2 = axes[0, 1]
    ic_series['Rank_IC'].hist(bins=30, ax=ax2, edgecolor='black')
    ax2.axvline(x=0, color='red', linestyle='--')
    ax2.set_title(f'IC分布 (均值={ic_series["Rank_IC"].mean():.4f})')
    
    # 3. 分层累计收益
    ax3 = axes[1, 0]
    for col in quantile_returns.columns:
        quantile_returns[col].cumsum().plot(ax=ax3, label=col)
    ax3.set_title('各分层累计收益')
    ax3.legend()
    
    # 4. 各分层平均收益柱状图
    ax4 = axes[1, 1]
    quantile_returns.mean().plot(kind='bar', ax=ax4, color='steelblue')
    ax4.set_title('各分层平均收益')
    ax4.axhline(y=0, color='red', linestyle='--')
    
    plt.tight_layout()
    plt.savefig('factor_analysis.png', dpi=150)
    plt.show()
```

## 五、总结

```
因子评价维度：

1. 预测能力    → IC / Rank IC / ICIR
2. 稳定性      → IC标准差 / IC胜率 / 滚动ICIR  
3. 单调性      → 分层回测（Q1~Q5收益是否单调递增）
4. 统计显著性  → t检验 / p值
5. 可交易性    → 因子自相关（换手率）/ 容量
6. 时效性      → 因子IC衰减曲线
```
