## 六、实盘联调命令

# 激活环境
.venv\Scripts\activate

# 单日信号（与回测同路径）
python -c "
import asyncio
from zstock.common.utils.db_utils import init_zstock_database, close_zstock_database
from zstock.strategy_management.signal_generator import SignalGenerator

async def main():
    await init_zstock_database()
    gen = SignalGenerator()
    df = await gen.generate_signals('2026-08-22')
    buys = df[df['signal_type']=='buy'] if 'signal_type' in df.columns else df.head(5)
    print(buys[['code','final_score','market_grade','position_scale']].to_string())
    await close_zstock_database()

asyncio.run(main())
"
